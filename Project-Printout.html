# Project Guidance

- This is a Windows desktop application written in C# with WinForms, targeting .NET 10.
- The current UI can select an EPANET 2.2 `.inp` file, read core node/link sections for a network summary, collect total demand, flow units, and per-zone instructions, validate them, calculate per-junction Unit-Length or Equal-Distribution demands, write a separate generated `.inp` copy, and reread/verify its junction base demands.
- Keep UI code separate from EPANET parsing, validation, and demand-distribution logic. Keep each future distribution method modular.
- Unit-Length requirements: known or metered demand is not used; each zone's entered demand is distributed using pipe lengths and K factors (default K = 1). Split each pipe's K-weighted length equally between its junction endpoints; a pipe connected to a reservoir or tank contributes its full K-weighted length to its junction endpoint. This makes same-zone pipes contribute their full length to the zone and cross-zone pipes contribute half to each zone. Normalize across all junctions assigned to each zone. A pipe with K = 0 contributes no weight to either endpoint; a junction may still receive weight from its other incident pipes.
- Equal-Distribution requirements: known or metered demand, pipe lengths, and K factors are not used. Within each zone, junctions default to weight 1; listed 50% junctions have weight 0.5, and listed 0% junctions have weight 0. Normalize weights across the zone to preserve its entered demand. A positive-demand zone cannot assign 0% to every junction.
- Preserve the source `.inp` file. Write generated output in the source folder with `_Unit-Length_Demand_Generated` or `_Equal-Distribution_Demand_Generated` in its name, according to the selected method; avoid overwriting existing output files. Replace junction base demands and remove active `[DEMANDS]` rows in the generated copy so prior demands are not added to the new allocation. Preserve demand patterns and unrelated EPANET data where practical, then reread and verify generated demand values.
- Validate all zone, junction, pipe, K-factor, and method-specific modifier instructions before calculating. Do not silently repair invalid input.
- Preserve unrelated EPANET data and formatting when writing generated files where practical, then reread and verify generated demand values.
- Do not add an EPANET simulation dependency unless needed and agreed; the current app prepares input files rather than running hydraulic analysis.
- Build with `dotnet build .\Demand_Gen.csproj` from the workspace root. Run on Windows with `dotnet run --project .\Demand_Gen.csproj`.

using System.Globalization;
using System.Text;

namespace Demand_Gen;

// Builds the human-readable audit report that accompanies a generated EPANET input file.
public static class DemandGenerationReportWriter
{
    /// <summary>
    /// Writes a detailed report of the source network, user instructions, allocation, and verification.
    /// </summary>
    public static string Write(
        string sourcePath,
        string generatedFilePath,
        EpanetNetworkSummary network,
        DemandValidationResult validation,
        DemandDistributionResult distribution)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(sourcePath);
        ArgumentException.ThrowIfNullOrWhiteSpace(generatedFilePath);
        ArgumentNullException.ThrowIfNull(network);
        ArgumentNullException.ThrowIfNull(validation);
        ArgumentNullException.ThrowIfNull(distribution);

        var reportPath = GetAvailableReportPath(generatedFilePath);
        var lengthUnit = GetLengthUnit(validation.FlowUnits);
        var totalPipeLength = network.Pipes.Sum(pipe => pipe.Length);
        var junctions = distribution.Zones.SelectMany(zone => zone.Junctions).ToArray();
        var totalCalculatedDemand = junctions.Sum(junction => junction.Demand);
        var junctionsWithDemand = junctions.Count(junction => junction.Demand > 0);
        var junctionsWithZeroDemand = junctions.Length - junctionsWithDemand;
        var report = new StringBuilder();

        // Record file names and generation time so the report can be matched to its input and output.
        report.AppendLine("EPANET Demand Generation Report");
        report.AppendLine($"Network name: {Path.GetFileNameWithoutExtension(sourcePath)}");
        report.AppendLine($"Original .inp file: {Path.GetFileName(sourcePath)}");
        report.AppendLine($"Revised .inp file: {Path.GetFileName(generatedFilePath)}");
        report.AppendLine($"Report file: {Path.GetFileName(reportPath)}");
        report.AppendLine($"Generated: {DateTimeOffset.Now.ToString("yyyy-MM-dd HH:mm:ss zzz", CultureInfo.InvariantCulture)}");
        report.AppendLine();

        report.AppendLine("Network summary");
        report.AppendLine($"Junctions: {network.JunctionIds.Count}");
        report.AppendLine($"Reservoirs: {network.ReservoirIds.Count}");
        report.AppendLine($"Tanks: {network.TankIds.Count}");
        report.AppendLine($"Total nodes: {network.JunctionIds.Count + network.ReservoirIds.Count + network.TankIds.Count}");
        report.AppendLine($"Pipes: {network.Pipes.Count}");
        report.AppendLine($"Pumps: {network.PumpCount}");
        report.AppendLine($"Valves: {network.ValveCount}");
        report.AppendLine($"Total links: {network.Pipes.Count + network.PumpCount + network.ValveCount}");
        report.AppendLine($"Total pipeline length: {FormatNumber(totalPipeLength)} {lengthUnit}");
        report.AppendLine($"Flow units: {validation.FlowUnits ?? "unknown"}");
        report.AppendLine();

        report.AppendLine("User-supplied data");
        report.AppendLine($"Distribution method: {GetMethodName(validation.Method)}");
        report.AppendLine($"Total demand entered: {FormatNumber(validation.TotalDemand.GetValueOrDefault())} {validation.FlowUnits ?? "unknown"}");
        report.AppendLine($"Number of zones: {validation.Zones.Count}");
        report.AppendLine(validation.Method == DemandDistributionMethod.UnitLength
            ? "Unlisted pipes use K = 1; K-weighted pipe lengths determine allocation."
            : "Default junction weight is 1; listed junction modifiers use weights 0.5 or 0.");
        foreach (var zone in validation.Zones)
        {
            report.AppendLine();
            report.AppendLine($"{zone.Name}");
            report.AppendLine($"  Demand entered: {FormatNumber(zone.Demand.GetValueOrDefault())} {validation.FlowUnits ?? "unknown"}");
            report.AppendLine($"  Junctions ({zone.Junctions.Count}): {DemandValidationResult.FormatIds(zone.Junctions)}");
            report.AppendLine(validation.Method == DemandDistributionMethod.UnitLength
                ? $"  K = 0.5 pipes: {DemandValidationResult.FormatIds(zone.KHalfPipes)}"
                : $"  50% junctions: {DemandValidationResult.FormatIds(zone.KHalfPipes)}");
            report.AppendLine(validation.Method == DemandDistributionMethod.UnitLength
                ? $"  K = 0 pipes: {DemandValidationResult.FormatIds(zone.KZeroPipes)}"
                : $"  0% junctions: {DemandValidationResult.FormatIds(zone.KZeroPipes)}");
        }

        report.AppendLine();
        report.AppendLine("Calculated results");
        report.AppendLine($"Total demand: {FormatNumber(totalCalculatedDemand)} {validation.FlowUnits ?? "unknown"}");
        report.AppendLine($"Junctions with demand: {junctionsWithDemand}");
        report.AppendLine($"Junctions with zero demand: {junctionsWithZeroDemand}");
        report.AppendLine();
        report.AppendLine("Output verification");
        // Compare per-zone and network totals independently to make allocation discrepancies visible.
        report.AppendLine($"Junction base demands verified: {junctions.Length} of {junctions.Length}");
        var enteredTotal = validation.TotalDemand.GetValueOrDefault();
        report.AppendLine($"Total demand entered: {FormatNumber(enteredTotal)} {validation.FlowUnits ?? "unknown"}");
        report.AppendLine($"Total demand allocated: {FormatNumber(totalCalculatedDemand)} {validation.FlowUnits ?? "unknown"}");
        report.AppendLine($"Allocation difference (allocated - entered): {FormatNumber(totalCalculatedDemand - enteredTotal)} {validation.FlowUnits ?? "unknown"}");
        foreach (var zone in distribution.Zones)
        {
            var zoneTotal = zone.Junctions.Sum(junction => junction.Demand);
            report.AppendLine();
            report.AppendLine($"{zone.ZoneName}: entered {FormatNumber(zone.RequestedDemand)} {validation.FlowUnits ?? "unknown"}; " +
            $"allocated {FormatNumber(zoneTotal)} {validation.FlowUnits ?? "unknown"}; " +
            $"difference {FormatNumber(zoneTotal - zone.RequestedDemand)} {validation.FlowUnits ?? "unknown"}; " +
            $"junctions {zone.Junctions.Count}");
            report.AppendLine(validation.Method == DemandDistributionMethod.UnitLength
                ? $"  Junction | Demand | K-weighted pipe length ({lengthUnit})"
                : "  Junction | Demand | Allocation weight");
            foreach (var junction in zone.Junctions)
            {
                report.AppendLine(
                    $"  {junction.JunctionId} | {FormatNumber(junction.Demand)} {validation.FlowUnits ?? "unknown"} | " +
                    FormatNumber(junction.AllocationWeight));
            }
        }

        if (validation.Warnings.Count > 0)
        {
            report.AppendLine();
            report.AppendLine("Warnings");
            foreach (var warning in validation.Warnings)
            {
                report.AppendLine($"- {warning}");
            }
        }

        report.AppendLine();
        report.AppendLine("The source network file was not modified.");
        report.AppendLine("Junction base demands in the revised file replace prior base demands; active [DEMANDS] records are removed and demand patterns are retained.");

        File.WriteAllText(reportPath, report.ToString(), new UTF8Encoding(encoderShouldEmitUTF8Identifier: false));
        return reportPath;
    }

    private static string GetAvailableReportPath(string generatedFilePath)
    {
        var basePath = Path.ChangeExtension(generatedFilePath, ".txt");
        if (!File.Exists(basePath))
        {
            return basePath;
        }

        var directory = Path.GetDirectoryName(basePath) ?? string.Empty;
        var fileName = Path.GetFileNameWithoutExtension(basePath);
        var sequence = 2;
        var candidate = Path.Combine(directory, $"{fileName}_{sequence}.txt");
        // Preserve earlier reports by choosing the next unused sequence number.
        while (File.Exists(candidate))
        {
            sequence++;
            candidate = Path.Combine(directory, $"{fileName}_{sequence}.txt");
        }

        return candidate;
    }

    private static string GetMethodName(DemandDistributionMethod method) => method switch
    {
        DemandDistributionMethod.UnitLength => "Unit-Length",
        DemandDistributionMethod.EqualDistribution => "Equal-Distribution",
        _ => throw new InvalidOperationException("The selected distribution method is not supported.")
    };

    private static string GetLengthUnit(string? flowUnits) => flowUnits?.ToUpperInvariant() switch
    {
        "CFS" or "GPM" or "MGD" or "IMGD" or "AFD" => "ft",
        "LPS" or "LPM" or "MLD" or "CMH" or "CMD" => "m",
        _ => "network length units"
    };

    private static string FormatNumber(double value) =>
        value.ToString("F4", CultureInfo.InvariantCulture);
}

using System.Globalization;
using System.Text;
using System.Text.RegularExpressions;

namespace Demand_Gen;

// Describes which weighting rules the validator and distributor should apply.
public enum DemandDistributionMethod
{
    UnitLength,
    EqualDistribution
}

// Raw grid values stay as strings until validation so invalid user input can be reported precisely.
public sealed record ZoneInstructionDraft(
    string? Name,
    string? Demand,
    string? Junctions,
    string? KHalfPipes,
    string? KZeroPipes);

// Contains normalized IDs and parsed values used after validation succeeds.
public sealed record ParsedZoneInstruction(
    string Name,
    double? Demand,
    IReadOnlyList<string> Junctions,
    IReadOnlyList<string> KHalfPipes,
    IReadOnlyList<string> KZeroPipes);

public sealed class DemandValidationResult
{
    internal DemandValidationResult(
        DemandDistributionMethod method,
        double? totalDemand,
        string? flowUnits,
        IReadOnlyList<ParsedZoneInstruction> zones,
        IReadOnlyList<string> errors,
        IReadOnlyList<string> warnings)
    {
        Method = method;
        TotalDemand = totalDemand;
        FlowUnits = flowUnits;
        Zones = zones;
        Errors = errors;
        Warnings = warnings;
    }

    public DemandDistributionMethod Method { get; }
    public double? TotalDemand { get; }
    public string? FlowUnits { get; }
    public IReadOnlyList<ParsedZoneInstruction> Zones { get; }
    public IReadOnlyList<string> Errors { get; }
    public IReadOnlyList<string> Warnings { get; }
    public bool IsValid => Errors.Count == 0;

    // Renders instructions, zone-specific issues, cross-zone issues, and warnings for the UI.
    public string BuildReport()
    {
        var report = new StringBuilder();
        report.AppendLine("Parsed instructions");
        var isUnitLength = Method == DemandDistributionMethod.UnitLength;
        var flowUnits = FlowUnits ?? "(no unit selected)";
        report.AppendLine($"Total demand: {FormatNumber(TotalDemand)} {flowUnits}");

        foreach (var zone in Zones)
        {
            var halfModifier = isUnitLength ? "K = 0.5 pipes" : "50% junctions";
            var zeroModifier = isUnitLength ? "K = 0 pipes" : "0% junctions";
            report.AppendLine(
                $"{zone.Name}: demand {FormatNumber(zone.Demand)} {flowUnits} | " +
                $"Junctions ({zone.Junctions.Count}): {FormatIds(zone.Junctions)} | " +
                $"{halfModifier}: {FormatIds(zone.KHalfPipes)} | {zeroModifier}: {FormatIds(zone.KZeroPipes)}");
        }

        report.AppendLine();
        report.AppendLine("Validation summary");
        report.AppendLine(IsValid
            ? "Status: passed. No demands have been calculated or written."
            : $"Status: {Errors.Count} issue(s) need correction.");
        if (!IsValid)
        {
            report.AppendLine($"Issues to fix ({Errors.Count})");
            var remainingErrors = Errors.ToList();

            // Group messages by zone first; keep unmatched messages in a network-wide section.
            foreach (var zone in Zones)
            {
                var zoneErrors = remainingErrors
                    .Where(error => IsZoneIssue(error, zone.Name))
                    .ToArray();
                if (zoneErrors.Length == 0)
                {
                    continue;
                }

                report.AppendLine($"{zone.Name}:");
                foreach (var error in zoneErrors)
                {
                    report.AppendLine($"- {error}");
                    remainingErrors.Remove(error);
                }
            }

            if (remainingErrors.Count > 0)
            {
                report.AppendLine("Network-wide or cross-zone:");
                foreach (var error in remainingErrors)
                {
                    report.AppendLine($"- {error}");
                }
            }
        }

        if (Warnings.Count > 0)
        {
            report.AppendLine();
            report.AppendLine($"Warnings ({Warnings.Count})");
            foreach (var warning in Warnings)
            {
                report.AppendLine($"- {warning}");
            }
        }

        return report.ToString();
    }

    private static bool IsZoneIssue(string error, string zoneName) =>
        error.StartsWith($"{zoneName} ", StringComparison.OrdinalIgnoreCase)
        || error.StartsWith($"{zoneName}:", StringComparison.OrdinalIgnoreCase)
        || error.Contains($" under {zoneName}", StringComparison.OrdinalIgnoreCase)
        || error.Contains($" in {zoneName};", StringComparison.OrdinalIgnoreCase);

    private static string FormatNumber(double? value) =>
        value?.ToString("F4", CultureInfo.InvariantCulture) ?? "invalid";

    internal static string FormatIds(IReadOnlyList<string> ids)
    {
        if (ids.Count == 0)
        {
            return "none";
        }

        var formattedIds = new List<string>();
        var index = 0;
        while (index < ids.Count)
        {
            // Compress only consecutive numeric IDs that share an identical prefix.
            var startMatch = Regex.Match(ids[index], "^(?<prefix>.*?)(?<number>\\d+)$");
            if (!startMatch.Success
                || !int.TryParse(startMatch.Groups["number"].Value, NumberStyles.None, CultureInfo.InvariantCulture, out var startNumber))
            {
                formattedIds.Add(ids[index++]);
                continue;
            }

            var prefix = startMatch.Groups["prefix"].Value;
            var endIndex = index;
            var previousNumber = startNumber;
            while (endIndex + 1 < ids.Count)
            {
                var nextMatch = Regex.Match(ids[endIndex + 1], "^(?<prefix>.*?)(?<number>\\d+)$");
                if (!nextMatch.Success
                    || !prefix.Equals(nextMatch.Groups["prefix"].Value, StringComparison.OrdinalIgnoreCase)
                    || !int.TryParse(nextMatch.Groups["number"].Value, NumberStyles.None, CultureInfo.InvariantCulture, out var nextNumber)
                    || (long)nextNumber != (long)previousNumber + 1)
                {
                    break;
                }

                endIndex++;
                previousNumber = nextNumber;
            }

            formattedIds.Add(endIndex == index ? ids[index] : $"{ids[index]}-{ids[endIndex]}");
            index = endIndex + 1;
        }

        return string.Join(", ", formattedIds);
    }
}

public static class DemandInputValidator
{
    public const double DemandMismatchToleranceFraction = 0.001;
    private static readonly Regex ZoneNamePattern = new("^[A-Za-z][A-Za-z0-9_-]{0,30}$", RegexOptions.Compiled);
    private static readonly Regex RangePattern = new("^(?<startPrefix>.*?)(?<startNumber>\\d+)-(?<endPrefix>.*?)(?<endNumber>\\d+)$", RegexOptions.Compiled);
    private static readonly string[] FlowUnits = ["CFS", "GPM", "MGD", "IMGD", "AFD", "LPS", "LPM", "MLD", "CMH", "CMD"];

    /// <summary>
    /// Parses and checks network-wide and zone-specific input without calculating or writing demands.
    /// </summary>
    public static DemandValidationResult Validate(
        EpanetNetworkSummary network,
        DemandDistributionMethod method,
        string? totalDemandText,
        string? flowUnits,
        IReadOnlyList<ZoneInstructionDraft> drafts)
    {
        var errors = new List<string>();
        var warnings = new List<string>();
        var parsedZones = new List<ParsedZoneInstruction>();
        var junctionIds = network.JunctionIds.ToDictionary(id => id, StringComparer.OrdinalIgnoreCase);
        var pipesById = network.Pipes.ToDictionary(pipe => pipe.Id, StringComparer.OrdinalIgnoreCase);
        var pipeIds = pipesById.Keys.ToDictionary(id => id, StringComparer.OrdinalIgnoreCase);
        var modifierIds = method == DemandDistributionMethod.UnitLength ? pipeIds : junctionIds;
        var isUnitLength = method == DemandDistributionMethod.UnitLength;
        var zoneNames = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        var junctionOwners = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
        var pipeAssignments = new Dictionary<string, (double K, string Zone)>(StringComparer.OrdinalIgnoreCase);
        var totalDemand = ParseNonNegativeNumber(totalDemandText, "Total demand", errors);

        // The selected units must be supported and must agree with the source network when it declares units.
        if (flowUnits is null || !FlowUnits.Contains(flowUnits, StringComparer.OrdinalIgnoreCase))
        {
            errors.Add("Select a valid EPANET flow unit.");
        }
        else if (network.FlowUnits is not null && !flowUnits.Equals(network.FlowUnits, StringComparison.OrdinalIgnoreCase))
        {
            errors.Add($"Selected flow unit '{flowUnits}' does not match the input network's flow unit '{network.FlowUnits}'.");
        }

        if (drafts.Count == 0)
        {
            errors.Add("At least one zone is required.");
        }

        var parsedDemands = new List<double>();
        foreach (var draft in drafts)
        {
            // Validate and normalize each row while accumulating ownership and modifier conflicts.
            var fallbackName = $"Zone row {parsedZones.Count + 1}";
            var zoneName = draft.Name?.Trim() ?? string.Empty;
            if (!ZoneNamePattern.IsMatch(zoneName))
            {
                errors.Add($"{fallbackName}: name must start with a letter and contain only letters, numbers, underscores, or hyphens (up to 31 characters).");
                zoneName = fallbackName;
            }
            else if (!zoneNames.Add(zoneName))
            {
                errors.Add($"Zone name '{zoneName}' is repeated.");
            }

            var zoneLabel = zoneName;
            var zoneDemand = ParseNonNegativeNumber(draft.Demand, $"{zoneLabel} demand", errors);
            if (zoneDemand.HasValue)
            {
                parsedDemands.Add(zoneDemand.Value);
            }

            var zoneJunctions = ParseIdList(
                draft.Junctions,
                $"{zoneLabel} junctions",
                junctionIds,
                required: true,
                errors);
            foreach (var junctionId in zoneJunctions)
            {
                if (!junctionOwners.TryAdd(junctionId, zoneLabel))
                {
                    errors.Add($"Junction '{junctionId}' is listed more than once, in {junctionOwners[junctionId]} and {zoneLabel}.");
                }
            }

            var kHalfPipes = ParseIdList(
                draft.KHalfPipes,
                isUnitLength ? $"{zoneLabel} K = 0.5 pipes" : $"{zoneLabel} 50% junctions",
                modifierIds,
                required: false,
                errors);
            var kZeroPipes = ParseIdList(
                draft.KZeroPipes,
                isUnitLength ? $"{zoneLabel} K = 0 pipes" : $"{zoneLabel} 0% junctions",
                modifierIds,
                required: false,
                errors);

            var halfPipeSet = new HashSet<string>(kHalfPipes, StringComparer.OrdinalIgnoreCase);
            foreach (var pipeId in kZeroPipes)
            {
                if (halfPipeSet.Contains(pipeId))
                {
                    errors.Add(isUnitLength
                        ? $"Pipe '{pipeId}' in {zoneLabel} is assigned both K = 0.5 and K = 0."
                        : $"Junction '{pipeId}' in {zoneLabel} is assigned both 50% and 0% demand.");
                }
            }

            if (isUnitLength)
            {
                AddPipeAssignments(kHalfPipes, 0.5, zoneLabel, pipeAssignments, errors);
                AddPipeAssignments(kZeroPipes, 0.0, zoneLabel, pipeAssignments, errors);
            }

            parsedZones.Add(new ParsedZoneInstruction(
                zoneLabel,
                zoneDemand,
                zoneJunctions,
                kHalfPipes,
                kZeroPipes));
        }

        if (parsedDemands.Count == drafts.Count && totalDemand.HasValue)
        {
            // A mismatch beyond 0.1% is an error; smaller non-rounding differences are shown as warnings.
            var zoneDemandTotal = parsedDemands.Sum();
            var difference = Math.Abs(zoneDemandTotal - totalDemand.Value);
            var tolerance = Math.Abs(totalDemand.Value) * DemandMismatchToleranceFraction;
            var roundingTolerance = Math.Max(1.0, Math.Abs(totalDemand.Value)) * 1e-9;
            if (difference > tolerance)
            {
                errors.Add($"Zone demands total {zoneDemandTotal.ToString("G", CultureInfo.InvariantCulture)}, but total demand is {totalDemand.Value.ToString("G", CultureInfo.InvariantCulture)}.");
            }
            else if (difference > roundingTolerance)
            {
                warnings.Add(
                    $"Zone totals differ from the Network total by {difference.ToString("F3", CultureInfo.InvariantCulture)} {flowUnits ?? "unknown"}.");
            }
        }

        foreach (var zone in parsedZones)
        {
            // A modifier must refer to an element assigned to (or incident to) the zone where it is entered.
            foreach (var modifierId in zone.KHalfPipes.Concat(zone.KZeroPipes))
            {
                if (isUnitLength)
                {
                    var pipe = pipesById[modifierId];
                    var connectsToZone =
                        junctionOwners.TryGetValue(pipe.StartNodeId, out var startOwner)
                            && startOwner.Equals(zone.Name, StringComparison.OrdinalIgnoreCase)
                        || junctionOwners.TryGetValue(pipe.EndNodeId, out var endOwner)
                            && endOwner.Equals(zone.Name, StringComparison.OrdinalIgnoreCase);

                    if (!connectsToZone)
                    {
                        errors.Add($"Pipe '{modifierId}' is listed under {zone.Name}, but neither endpoint junction belongs to that zone.");
                    }
                }
                else if (!junctionOwners.TryGetValue(modifierId, out var owner)
                    || !owner.Equals(zone.Name, StringComparison.OrdinalIgnoreCase))
                {
                    errors.Add($"Junction '{modifierId}' is listed as a demand modifier under {zone.Name}, but it is not assigned to that zone.");
                }
            }

            if (isUnitLength && zone.Demand.GetValueOrDefault() > 0)
            {
                // Mirror the distributor's endpoint split to reject zones that cannot receive positive weight.
                var totalPipeWeight = 0.0;
                foreach (var pipe in network.Pipes)
                {
                    var k = pipeAssignments.TryGetValue(pipe.Id, out var assignment) ? assignment.K : 1.0;
                    var effectiveLength = pipe.Length * k;
                    var hasStartJunction = junctionOwners.TryGetValue(pipe.StartNodeId, out var startOwner);
                    var hasEndJunction = junctionOwners.TryGetValue(pipe.EndNodeId, out var endOwner);
                    var endpointWeight = hasStartJunction && hasEndJunction
                        ? effectiveLength / 2.0
                        : effectiveLength;

                    if (hasStartJunction && startOwner!.Equals(zone.Name, StringComparison.OrdinalIgnoreCase))
                    {
                        totalPipeWeight += endpointWeight;
                    }

                    if (hasEndJunction && endOwner!.Equals(zone.Name, StringComparison.OrdinalIgnoreCase))
                    {
                        totalPipeWeight += endpointWeight;
                    }
                }

                if (!double.IsFinite(totalPipeWeight))
                {
                    errors.Add($"{zone.Name} has a non-finite total K-weighted pipe length.");
                }
                else if (totalPipeWeight <= 0)
                {
                    errors.Add($"{zone.Name} has positive demand, but no positive K-weighted pipe length.");
                }
            }

            if (!isUnitLength
                && zone.Demand.GetValueOrDefault() > 0
                && zone.Junctions.Count > 0
                && zone.Junctions.All(id => zone.KZeroPipes.Contains(id, StringComparer.OrdinalIgnoreCase)))
            {
                errors.Add($"{zone.Name} has positive demand, but every junction is assigned 0% demand.");
            }
        }

        var omittedJunctions = network.JunctionIds.Where(id => !junctionOwners.ContainsKey(id)).ToArray();
        if (omittedJunctions.Length > 0)
        {
            errors.Add($"Junctions omitted from all zones: {DemandValidationResult.FormatIds(omittedJunctions)}.");
        }

        return new DemandValidationResult(method, totalDemand, flowUnits, parsedZones, errors, warnings);
    }

    private static double? ParseNonNegativeNumber(string? text, string fieldName, ICollection<string> errors)
    {
        if (!double.TryParse(text, NumberStyles.Float, CultureInfo.CurrentCulture, out var value)
            && !double.TryParse(text, NumberStyles.Float, CultureInfo.InvariantCulture, out value))
        {
            errors.Add($"{fieldName} must be a number.");
            return null;
        }

        if (!double.IsFinite(value) || value < 0)
        {
            errors.Add($"{fieldName} must be a finite, non-negative number.");
            return null;
        }

        return value;
    }

    private static IReadOnlyList<string> ParseIdList(
        string? text,
        string fieldName,
        IReadOnlyDictionary<string, string> availableIds,
        bool required,
        ICollection<string> errors)
    {
        if (string.IsNullOrWhiteSpace(text))
        {
            if (required)
            {
                errors.Add($"{fieldName} is required.");
            }

            return [];
        }

        var result = new List<string>();
        var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        // Commas separate IDs/ranges; an empty token is an error rather than something to silently ignore.
        var tokens = text.Split(',', StringSplitOptions.TrimEntries);
        foreach (var token in tokens)
        {
            if (token.Length == 0)
            {
                errors.Add($"{fieldName} contains an empty item.");
                continue;
            }

            if (!TryExpandToken(token, availableIds.Count, out var expandedIds, out var rangeError))
            {
                errors.Add($"{fieldName}: {rangeError}");
                continue;
            }

            foreach (var expandedId in expandedIds)
            {
                if (!availableIds.TryGetValue(expandedId, out var canonicalId))
                {
                    errors.Add($"{fieldName} refers to unknown ID '{expandedId}'.");
                    continue;
                }

                if (!seen.Add(canonicalId))
                {
                    errors.Add($"{fieldName} repeats ID '{canonicalId}'.");
                    continue;
                }

                result.Add(canonicalId);
            }
        }

        return result;
    }

    private static bool TryExpandToken(string token, int maximumCount, out IReadOnlyList<string> ids, out string error)
    {
        var match = RangePattern.Match(token);
        if (!match.Success)
        {
            ids = [token];
            error = string.Empty;
            return true;
        }

        var startPrefix = match.Groups["startPrefix"].Value;
        var endPrefix = match.Groups["endPrefix"].Value;
        if (!startPrefix.Equals(endPrefix, StringComparison.OrdinalIgnoreCase))
        {
            ids = [];
            error = $"range '{token}' must use the same prefix at both ends.";
            return false;
        }

        if (!int.TryParse(match.Groups["startNumber"].Value, NumberStyles.None, CultureInfo.InvariantCulture, out var start)
            || !int.TryParse(match.Groups["endNumber"].Value, NumberStyles.None, CultureInfo.InvariantCulture, out var end))
        {
            ids = [];
            error = $"range '{token}' contains an invalid number.";
            return false;
        }

        if (end < start)
        {
            ids = [];
            error = $"range '{token}' ends before it starts.";
            return false;
        }

        var count = (long)end - start + 1;
        if (count > Math.Max(maximumCount, 1))
        {
            ids = [];
            error = $"range '{token}' expands beyond the number of items in the network.";
            return false;
        }

        var width = match.Groups["startNumber"].Length;
        // Retain zero padding from the first ID (for example, J01-J03 expands to J01, J02, J03).
        ids = Enumerable.Range(start, (int)count)
            .Select(number => $"{startPrefix}{number.ToString(CultureInfo.InvariantCulture).PadLeft(width, '0')}")
            .ToArray();
        error = string.Empty;
        return true;
    }

    private static void AddPipeAssignments(
        IEnumerable<string> pipeIds,
        double k,
        string zone,
        IDictionary<string, (double K, string Zone)> assignments,
        ICollection<string> errors)
    {
        foreach (var pipeId in pipeIds)
        {
            // A pipe can have one K value only, even when it connects two different zones.
            if (assignments.TryGetValue(pipeId, out var existing))
            {
                errors.Add($"Pipe '{pipeId}' has a repeated K assignment in {zone}; it is already assigned K = {existing.K.ToString("G", CultureInfo.InvariantCulture)} in {existing.Zone}.");
                continue;
            }

            assignments.Add(pipeId, (k, zone));
        }
    }
}

using System.Globalization;
using System.Text;
using System.Text.RegularExpressions;

namespace Demand_Gen;

public static class EpanetDemandFileWriter
{
    private static readonly Regex TokenPattern = new("\\S+", RegexOptions.Compiled);
    private static readonly Regex LineBreakPattern = new("\\r\\n|\\n|\\r", RegexOptions.Compiled);

    /// <summary>
    /// Creates a new input copy, then rereads it and verifies each junction demand before publishing it.
    /// </summary>
    public static string WriteVerifiedCopy(string sourcePath, DemandDistributionResult distribution)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(sourcePath);
        ArgumentNullException.ThrowIfNull(distribution);

        var sourceFullPath = Path.GetFullPath(sourcePath);
        var outputPath = GetAvailableOutputPath(sourceFullPath, distribution.Method);
        var temporaryPath = $"{outputPath}.{Guid.NewGuid():N}.tmp";
        var expectedDemands = CreateDemandMap(distribution);

        try
        {
            // Write to a unique temporary file so a failed update never leaves a partial final output.
            var source = ReadSourceText(sourceFullPath);
            var outputText = ReplaceDemands(source.Content, expectedDemands);
            WriteText(temporaryPath, outputText, source.Encoding, source.HasPreamble);

            // Validate the generated file structure and values before moving it to its final name.
            EpanetNetworkReader.Read(temporaryPath);
            VerifyDemands(temporaryPath, expectedDemands);
            File.Move(temporaryPath, outputPath);
            return outputPath;
        }
        finally
        {
            if (File.Exists(temporaryPath))
            {
                File.Delete(temporaryPath);
            }
        }
    }

    private static Dictionary<string, double> CreateDemandMap(DemandDistributionResult distribution)
    {
        // Build a case-insensitive lookup and reject duplicate or invalid allocations before editing text.
        var demands = new Dictionary<string, double>(StringComparer.OrdinalIgnoreCase);
        foreach (var junction in distribution.Zones.SelectMany(zone => zone.Junctions))
        {
            if (!double.IsFinite(junction.Demand) || junction.Demand < 0)
            {
                throw new InvalidDataException($"Junction '{junction.JunctionId}' has an invalid calculated demand.");
            }

            if (!demands.TryAdd(junction.JunctionId, junction.Demand))
            {
                throw new InvalidDataException($"Junction '{junction.JunctionId}' has more than one calculated demand.");
            }
        }

        if (demands.Count == 0)
        {
            throw new InvalidDataException("The calculation contains no junction demands to write.");
        }

        return demands;
    }

    private static string ReplaceDemands(string content, IReadOnlyDictionary<string, double> expectedDemands)
    {
        var newline = content.Contains("\r\n", StringComparison.Ordinal)
            ? "\r\n"
            : content.Contains('\n') ? "\n" : "\r";
        var lines = LineBreakPattern.Split(content);
        var outputLines = new List<string>(lines.Length);
        var updatedJunctions = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        var section = string.Empty;

        foreach (var originalLine in lines)
        {
            var trimmedLine = originalLine.Trim();
            if (trimmedLine.StartsWith('[') && trimmedLine.Contains(']'))
            {
                section = trimmedLine[1..trimmedLine.IndexOf(']')].Trim().ToUpperInvariant();
                outputLines.Add(originalLine);
                continue;
            }

            if (section.Equals("DEMANDS", StringComparison.OrdinalIgnoreCase)
                && trimmedLine.Length > 0
                && !trimmedLine.StartsWith(';'))
            {
                // Remove active extra demands to prevent them being added on top of the new base demands.
                // Keep inline comments as standalone lines so user notes are not lost.
                var commentIndex = originalLine.IndexOf(';');
                if (commentIndex >= 0)
                {
                    outputLines.Add(originalLine[commentIndex..]);
                }

                continue;
            }

            if (!section.Equals("JUNCTIONS", StringComparison.OrdinalIgnoreCase)
                || trimmedLine.Length == 0
                || trimmedLine.StartsWith(';'))
            {
                outputLines.Add(originalLine);
                continue;
            }

            var commentStart = originalLine.IndexOf(';');
            var dataLength = commentStart >= 0 ? commentStart : originalLine.Length;
            var tokens = TokenPattern.Matches(originalLine[..dataLength]);
            if (tokens.Count < 2)
            {
                throw new InvalidDataException("A junction record is missing its ID or elevation.");
            }

            var junctionId = tokens[0].Value;
            if (!expectedDemands.TryGetValue(junctionId, out var demand))
            {
                throw new InvalidDataException($"Junction '{junctionId}' has no calculated demand.");
            }

            var formattedDemand = demand.ToString("G17", CultureInfo.InvariantCulture);
            if (tokens.Count >= 3)
            {
                // Replace only the demand token; leave spacing, optional pattern, and inline comment intact.
                var demandToken = tokens[2];
                outputLines.Add(originalLine.Remove(demandToken.Index, demandToken.Length)
                    .Insert(demandToken.Index, formattedDemand));
            }
            else
            {
                // EPANET permits junctions with no base-demand field, so insert one after elevation.
                var elevationToken = tokens[1];
                outputLines.Add(originalLine.Insert(elevationToken.Index + elevationToken.Length, $" {formattedDemand}"));
            }

            updatedJunctions.Add(junctionId);
        }

        var missingJunctions = expectedDemands.Keys.Where(id => !updatedJunctions.Contains(id)).ToArray();
        if (missingJunctions.Length > 0)
        {
            throw new InvalidDataException(
                $"Calculated junctions were not found in the source [JUNCTIONS] section: {string.Join(", ", missingJunctions)}.");
        }

        return string.Join(newline, outputLines);
    }

    private static void VerifyDemands(string filePath, IReadOnlyDictionary<string, double> expectedDemands)
    {
        var actualDemands = ReadJunctionDemands(filePath);
        foreach (var (junctionId, expectedDemand) in expectedDemands)
        {
            // Allow only a tiny floating-point tolerance after writing with round-trip precision.
            if (!actualDemands.TryGetValue(junctionId, out var actualDemand)
                || Math.Abs(actualDemand - expectedDemand) > Math.Max(1.0, Math.Abs(expectedDemand)) * 1e-12)
            {
                throw new InvalidDataException($"Generated demand verification failed for junction '{junctionId}'.");
            }
        }
    }

    private static Dictionary<string, double> ReadJunctionDemands(string filePath)
    {
        var demands = new Dictionary<string, double>(StringComparer.OrdinalIgnoreCase);
        var section = string.Empty;

        foreach (var originalLine in File.ReadLines(filePath))
        {
            var trimmedLine = originalLine.Trim();
            if (trimmedLine.StartsWith('[') && trimmedLine.Contains(']'))
            {
                section = trimmedLine[1..trimmedLine.IndexOf(']')].Trim().ToUpperInvariant();
                continue;
            }

            if (!section.Equals("JUNCTIONS", StringComparison.OrdinalIgnoreCase)
                || trimmedLine.Length == 0
                || trimmedLine.StartsWith(';'))
            {
                continue;
            }

            var commentStart = originalLine.IndexOf(';');
            var dataLength = commentStart >= 0 ? commentStart : originalLine.Length;
            var tokens = TokenPattern.Matches(originalLine[..dataLength]);
            if (tokens.Count < 3
                || !double.TryParse(tokens[2].Value, NumberStyles.Float, CultureInfo.InvariantCulture, out var demand)
                || !double.IsFinite(demand)
                || !demands.TryAdd(tokens[0].Value, demand))
            {
                throw new InvalidDataException("The generated [JUNCTIONS] section contains an invalid or duplicate demand record.");
            }
        }

        return demands;
    }

    private static string GetAvailableOutputPath(string sourcePath, DemandDistributionMethod method)
    {
        var directory = Path.GetDirectoryName(sourcePath)
            ?? throw new InvalidDataException("Could not determine the source network folder.");
        var fileName = Path.GetFileNameWithoutExtension(sourcePath);
        var methodName = method switch
        {
            DemandDistributionMethod.UnitLength => "Unit-Length",
            DemandDistributionMethod.EqualDistribution => "Equal-Distribution",
            _ => throw new InvalidDataException("The selected distribution method is not supported.")
        };
        var sequence = 1;
        var candidate = Path.Combine(directory, $"{fileName}_{methodName}_{sequence}.inp");
        // Never overwrite an earlier generated file; advance the suffix until the name is free.
        while (File.Exists(candidate))
        {
            sequence++;
            candidate = Path.Combine(directory, $"{fileName}_{methodName}_{sequence}.inp");
        }

        return candidate;
    }

    private static SourceText ReadSourceText(string filePath)
    {
        // Detect encoding and byte-order mark so the generated copy retains the source text format.
        var bytes = File.ReadAllBytes(filePath);
        using var stream = new MemoryStream(bytes);
        using var reader = new StreamReader(stream, Encoding.UTF8, detectEncodingFromByteOrderMarks: true);
        var content = reader.ReadToEnd();
        var encoding = reader.CurrentEncoding;
        var preamble = encoding.GetPreamble();
        var hasPreamble = preamble.Length > 0 && bytes.AsSpan().StartsWith(preamble);
        return new SourceText(content, encoding, hasPreamble);
    }

    private static void WriteText(string filePath, string content, Encoding encoding, bool hasPreamble)
    {
        // CreateNew prevents an unexpected existing temporary file from being overwritten.
        using var stream = new FileStream(filePath, FileMode.CreateNew, FileAccess.Write, FileShare.None);
        if (hasPreamble)
        {
            stream.Write(encoding.GetPreamble());
        }

        stream.Write(encoding.GetBytes(content));
    }

    private sealed record SourceText(string Content, Encoding Encoding, bool HasPreamble);
}

## EPANET-JDA (EPANET Junction Demand Allocator)

## User Guide & Technical Reference Manual — Version 1.0.1 {#manual-subtitle}


### 1. Overview and Capabilities
EPANET-JDA is a Windows-based utility which calculates junction base demands of an EPANET 2.2 hydraulic network model. It supports two standard demand allocation methodologies: **Unit-Length** and **Equal-Distribution** (refer later sections for detail). The application operates strictly as a pre-processor; it does not execute EPANET hydraulic simulations and does not require the EPANET simulation engine installed on the host machine.

The program reads a user-supplied input (.inp) file of EPANET 2.2, computes junction demands according to user-defined zones and weighting parameters, and writes a separate, verified input (.inp) file. Existing junction base demands in the **[JUNCTIONS]** section are replaced with the calculated values. Any active records in the **[DEMANDS]** section are removed to eliminate double-counting. Associated demand pattern assignments, operational controls, curves, and link parameters remain untouched. The source file is never overwritten or altered.

### 2. System Requirements and Installation

**Operating System:** Windows 10 or Windows 11.
**Runtime:** Standalone desktop executable (no external installer or runtime setup wizard required).
**Package Archive (epanetjda.zip):** Contains epanetjda.exe, EPANET_JDA.pdf, and one EPANET 2.2 sample input (.inp) file. 
Extract the entire archive to any local directory.

### 3. Typical Workflow

3.1 **Export Network File:** In EPANET 2.2, open or verify your network. Navigate to **File** &gt; **Export** &gt; **Network...** and export a clean input (.inp) file.

3.2 **Determine Demand Boundaries:**
For **Unit-Length**, compile total zone demands, zone junction lists, and pipe IDs where K ≠ 1.0.
For **Equal-Distribution**, compile total zone demands, zone junction lists, and junction IDs where N ≠ 1.0.
    
  
3.3 **Launch the Program:** Open epanetjda.exe. The initial workspace appears as shown in **Figure 1**.

![Figure 1: Main window interface before selecting a network file](help-images/Figure-2.png)
Ensure that, when the Browse... button is highlighted (receives focus), a help message is displayed.

3.4 **Load .inp file:** Click **Browse...** to select your .inp file. Check the status panel to verify junction counts, link counts, and network flow units.

3.5 **Select Method:** Choose **Unit-Length** or **Equal-Distribution** from the method drop-down list.

3.6 **Set Demand &amp; Units:** Enter the total network demand and confirm that the selected flow unit strictly matches the unit configured in the source file's **[OPTIONS]** section.

3.7 **Define Zones:** Enter the number of zones; matching rows will automatically populate in the Zone Detail grid.

3.8 **Populate Zone Data:** Provide unique zone names, target zone demands, and assigned Junction IDs (using comma-separated values and/or numeric ranges like J1-J5, J7, J9).

3.9 **Assign Modifiers:**

3.9.1 Under **Unit-Length**, enter pipe IDs requiring K = 0.5 or K = 0.0.

3.9.2 Under **Equal-Distribution**, enter junction IDs requiring N = 0.5 or N = 0.0.

3.10 **Validate Inputs:** Click **Validate**. Correct any reported errors in the validation pane. Validation verifies network topology and demand balances without modifying files.

3.11 **Calculate &amp; Save:** Click **Calculate &amp; Save**. The program computes allocations, generates a new .inp file, and also saves an analysis report (.txt) in the same folder as the source file. The output filename appears in the window; make a note of it.

3.12 **File Versioning:** Generated files are automatically saved with the original name plus name of the method used adding sequential tags (e.g., Sample1_UnitLength_1.inp or Sample1_EqualDistribution_1.inp). Existing input files are never overwritten.

The application does not run a hydraulic simulation.

The complete build and run instructions are also available in the project README file. 

### 4. Input Rules and Validation Constraints

### 4.1 Network Total and Flow Units

Network total demand must be a finite, positive number.
Supported EPANET flow units: CFS, GPM, MGD, IMGD, AFD, LPS, LPM, MLD, CMH, and CMD.
The selected unit must match the unit declared in the network's **[OPTIONS]** section.
The sum of all zone demands must reconcile with the network total demand within a tolerance of 0.1%. Discrepancies exceeding 0.1% are flagged as fatal validation errors; minor rounding variances trigger informational warnings.

### 4.2 Demand Pattern and Section Handling

**[JUNCTIONS]** Section: The base demand column (Column 3) of the .inp file is updated with calculated values. If an existing junction entry references a Demand Pattern ID (Column 4), that pattern association remains preserved.
**[DEMANDS]** Section: Any secondary or extended demand records are removed to prevent duplicating consumer baseline drafts.
Original .inp: The original .inp file remains completely untouched.

### 4.3 Zone and Element Constraints

Zone identifiers must begin with an alphabetic character, may contain alphanumeric characters, hyphens, or underscores, and must not exceed 31 characters.
Every junction in the network must be assigned. Unassigned junctions or duplicate junction assignments between zones will trigger validation failures.
Reservoirs and tanks shares no demand and cannot be assigned to zones.
Non-pipe link elements (pumps, PRVs, PSVs, PBVs, FCVs, TCVs) possess no physical pipeline length and are omitted from Unit-Length weighting calculations.

### 5. Method-specific modifier columns

The last two grid columns have different meanings depending on the selected method.

| Method | Column 1 modifier | Column 2 modifier | Default (Unlisted) |
| :---: | --- | --- | --- |
| **Unit-Length** | Pipe IDs with K = 0.5 | Pipe IDs with K = 0 | K = 1.0 (Full weight) |
| **Equal-Distribution** | Junction IDs with N = 0.5 (50%) | Junction IDs with N = 0.0 (0%) | N = 1.0 (100% weight) |

For **Unit-Length**, unlisted pipes use K = 1. A pipe modifier must be entered under a zone that contains at least one endpoint of that pipe. A pipe can have only one K assignment. If a pipe belongs to two zones, its ID can be entered in either zone, but not both.

Every positive-demand Unit-Length zone must have positive K-weighted pipe length contributing to its junctions. A pipe with K = 0 contributes no weight to either endpoint.

For **Equal-Distribution**, unlisted junctions use 100% weight. A positive-demand zone must retain at least one junction with positive weight; it cannot assign 0% weight to every junction.

A valve or pump is a link without a pipe length and is not used by Unit-Length length weighting. For example, the valve `V1` in the sample network connects `J2` and `J6`; it has no length contribution. `J6` remains a normal junction assigned to Z2.

### 6. Calculation methods

### 6.1 Unit-Length

**Unit-Length** distributes each zone's entered demand according to K-weighted pipe-length contributions to its junctions. Known or metered junction demands are not used.

For each pipe `p`:

$$
L_{p,\mathrm{eff}} = K_p L_p
$$

For a junction endpoint `j` of pipe `p`, the pipe contribution is:

$$
w_{j,p} = \frac{K_p L_p}{n_J(p)}
$$

where `n_J(p)` is the number of junction endpoints. It is 2 for a junction-to-junction pipe and 1 when the other endpoint is a reservoir or tank. The junction weight is the sum of the contributions from all incident pipes:

$$
w_j = \sum_{p \text{ incident to } j} w_{j,p}
$$

A pipe connected to a reservoir or tank contributes its full effective length to the junction endpoint. A pipe between two junctions contributes half of its effective length to each endpoint. For zone `z`, the zone weight and junction demand are:

$$
W_z = \sum_{j \in J_z} w_j, \qquad
D_j = D_z \frac{w_j}{W_z}
$$

A positive-demand zone must have `W_z > 0`. The application adjusts floating-point rounding at one junction so the calculated demands sum to the entered zone demand.

### 6.2 Equal-Distribution

**Equal-Distribution** does not use pipe lengths, K factors, or known or metered junction demands. Junction weights are 1.0 for unlisted junctions, 0.5 for junctions listed at 50%, and 0 for junctions listed at 0%.

For each junction `j`, let `a_j` be its allocation weight. For zone `z`:

$$
A_z = \sum_{j \in J_z} a_j, \qquad
D_j = D_z \frac{a_j}{A_z}
$$

Calculated demands within each zone sum to the entered zone demand. A zero-demand zone can have zero weights throughout.

### 7. Worked example: sample network

The package includes `Sample1.inp`. It contains 10 junctions (`J1` through `J10`), one reservoir (`R1`), one valve (`V1` connecting J2 to J6), 10 pipes, and the flow unit `LPS` with total network demand defined at 50 LPS.

### 8. Reviewing generated files

The detailed `.txt` report lists the network summary, entered instructions, calculated junction demands and weights, warnings, and entered-versus-allocated totals. Check that the zone and network totals match the intended values.

If generation fails, read the validation or error text in the application, correct the source or entered values, validate again, and generate a new copy. Existing generated files and reports are preserved; a new sequence number is used rather than overwriting them.

public static class EpanetNetworkReader
{
    /// <summary>
    /// Reads the node, pipe, pump, valve, and flow-unit data needed by the allocator.
    /// </summary>
    public static EpanetNetworkSummary Read(string filePath)
    {
        var junctionIds = new List<string>();
        var reservoirIds = new List<string>();
        var tankIds = new List<string>();
        var pipes = new List<EpanetPipe>();
        var links = new List<(string Id, string StartNodeId, string EndNodeId)>();
        var nodeIds = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        var linkIds = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        var pumpCount = 0;
        var valveCount = 0;
        string? flowUnits = null;
        var section = string.Empty;
        var lines = File.ReadAllLines(filePath);

        for (var index = 0; index < lines.Length; index++)
        {
            var line = lines[index].Trim();
            // Blank lines and full-line comments do not contribute network records.
            if (line.Length == 0 || line.StartsWith(';'))
            {
                continue;
            }

            if (line.StartsWith('[') && line.Contains(']'))
            {
                section = line[1..line.IndexOf(']')].Trim().ToUpperInvariant();
                continue;
            }

            // Strip inline comments before splitting the EPANET data fields.
            var commentIndex = line.IndexOf(';');
            if (commentIndex >= 0)
            {
                line = line[..commentIndex].Trim();
            }

            if (line.Length == 0)
            {
                continue;
            }

            var fields = line.Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries);
            var lineNumber = index + 1;

            // Only parse the network sections needed for summaries and demand allocation.
            switch (section)
            {
                case "JUNCTIONS":
                    RequireFields(fields, 2, section, lineNumber);
                    AddUnique(fields[0], junctionIds, nodeIds, section, lineNumber);
                    break;
                case "RESERVOIRS":
                    RequireFields(fields, 2, section, lineNumber);
                    AddUnique(fields[0], reservoirIds, nodeIds, section, lineNumber);
                    break;
                case "TANKS":
                    RequireFields(fields, 6, section, lineNumber);
                    AddUnique(fields[0], tankIds, nodeIds, section, lineNumber);
                    break;
                case "PIPES":
                    RequireFields(fields, 6, section, lineNumber);
                    AddLink(fields[0], fields[1], fields[2], links, linkIds, section, lineNumber);
                    if (!double.TryParse(fields[3], NumberStyles.Float, CultureInfo.InvariantCulture, out var length)
                        || !double.IsFinite(length)
                        || length < 0)
                    {
                        throw InvalidRecord(section, lineNumber, "pipe length must be a non-negative number");
                    }

                    pipes.Add(new EpanetPipe(fields[0], fields[1], fields[2], length));
                    break;
                case "PUMPS":
                    RequireFields(fields, 4, section, lineNumber);
                    if (!fields.Skip(3).Any(field =>
                            field.Equals("POWER", StringComparison.OrdinalIgnoreCase)
                            || field.Equals("HEAD", StringComparison.OrdinalIgnoreCase)))
                    {
                        throw InvalidRecord(section, lineNumber, "pump must specify POWER or HEAD");
                    }

                    AddLink(fields[0], fields[1], fields[2], links, linkIds, section, lineNumber);
                    pumpCount++;
                    break;
                case "VALVES":
                    RequireFields(fields, 7, section, lineNumber);
                    AddLink(fields[0], fields[1], fields[2], links, linkIds, section, lineNumber);
                    valveCount++;
                    break;
                case "OPTIONS":
                    if (fields.Length >= 2 && fields[0].Equals("UNITS", StringComparison.OrdinalIgnoreCase))
                    {
                        flowUnits = fields[1];
                    }

                    break;
            }
        }

        if (junctionIds.Count == 0)
        {
            throw new InvalidDataException("The input file does not contain any junctions in its [JUNCTIONS] section.");
        }

        // Check every encountered link after all node sections have been read.
        foreach (var link in links)
        {
            if (!nodeIds.Contains(link.StartNodeId) || !nodeIds.Contains(link.EndNodeId))
            {
                throw new InvalidDataException(
                    $"Link '{link.Id}' refers to an undefined node ('{link.StartNodeId}' or '{link.EndNodeId}').");
            }
        }

        return new EpanetNetworkSummary(
            junctionIds.ToArray(),
            reservoirIds.ToArray(),
            tankIds.ToArray(),
            pipes.ToArray(),
            pumpCount,
            valveCount,
            flowUnits);
    }

    private static void AddUnique(
        string id,
        ICollection<string> ids,
        ISet<string> allIds,
        string section,
        int lineNumber)
    {
        // Node IDs share one namespace, so duplicates across node types are rejected as well.
        if (!allIds.Add(id))
        {
            throw InvalidRecord(section, lineNumber, $"duplicate node ID '{id}'");
        }

        ids.Add(id);
    }

    private static void AddLink(
        string id,
        string startNodeId,
        string endNodeId,
        ICollection<(string Id, string StartNodeId, string EndNodeId)> links,
        ISet<string> allIds,
        string section,
        int lineNumber)
    {
        // Link IDs are unique across pipes, pumps, and valves.
        if (!allIds.Add(id))
        {
            throw InvalidRecord(section, lineNumber, $"duplicate link ID '{id}'");
        }

        links.Add((id, startNodeId, endNodeId));
    }

    private static void RequireFields(string[] fields, int count, string section, int lineNumber)
    {
        if (fields.Length < count)
        {
            throw InvalidRecord(section, lineNumber, $"expected at least {count} fields but found {fields.Length}");
        }
    }

    private static InvalidDataException InvalidRecord(string section, int lineNumber, string message) =>
        new($"Invalid record in [{section}] on line {lineNumber}: {message}.");
}

namespace Demand_Gen;

// Allocates each zone evenly after applying the supported junction weight modifiers.
public static class EqualDistributionDemandDistributor
{
    /// <summary>
    /// Distributes demand using only junction modifiers; pipe lengths and K factors are ignored.
    /// </summary>
    public static DemandDistributionResult Calculate(DemandValidationResult validation)
    {
        ArgumentNullException.ThrowIfNull(validation);
        if (!validation.IsValid)
        {
            throw new InvalidOperationException("Demand instructions must pass validation before calculation.");
        }

        if (validation.Method != DemandDistributionMethod.EqualDistribution)
        {
            throw new InvalidOperationException("Equal-Distribution calculation requires Equal-Distribution validation.");
        }

        var zoneAllocations = new List<ZoneDemandAllocation>(validation.Zones.Count);
        foreach (var zone in validation.Zones)
        {
            var requestedDemand = zone.Demand
                ?? throw new InvalidOperationException($"Zone '{zone.Name}' has no validated demand.");

            // The input model reuses the K-named modifier fields for junction modifiers in this method.
            var halfDemandJunctions = new HashSet<string>(zone.KHalfPipes, StringComparer.OrdinalIgnoreCase);
            var excludedJunctions = new HashSet<string>(zone.KZeroPipes, StringComparer.OrdinalIgnoreCase);
            var weights = zone.Junctions.ToDictionary(
                id => id,
                id => excludedJunctions.Contains(id) ? 0.0 : halfDemandJunctions.Contains(id) ? 0.5 : 1.0,
                StringComparer.OrdinalIgnoreCase);
            var totalWeight = weights.Values.Sum();

            if (requestedDemand > 0 && totalWeight <= 0)
            {
                throw new InvalidOperationException(
                    $"Zone '{zone.Name}' has positive demand but no junction with positive allocation weight.");
            }

            var demands = zone.Junctions.ToDictionary(
                id => id,
                _ => 0.0,
                StringComparer.OrdinalIgnoreCase);
            if (requestedDemand > 0)
            {
                // Normalize each junction's weight so the zone receives exactly its requested total.
                foreach (var junctionId in zone.Junctions)
                {
                    demands[junctionId] = requestedDemand * (weights[junctionId] / totalWeight);
                }

                // Correct tiny floating-point residue on the junction with the greatest allocation weight.
                var allocated = zone.Junctions.Sum(id => demands[id]);
                var correctionJunction = zone.Junctions.MaxBy(id => weights[id]);
                demands[correctionJunction!] += requestedDemand - allocated;
            }

            var junctionAllocations = zone.Junctions
                .Select(id => new JunctionDemandAllocation(id, zone.Name, weights[id], demands[id]))
                .ToArray();
            zoneAllocations.Add(new ZoneDemandAllocation(zone.Name, requestedDemand, junctionAllocations));
        }

        return new DemandDistributionResult(DemandDistributionMethod.EqualDistribution, zoneAllocations);
    }
}

    private void CalculateButton_Click(object? sender, EventArgs e)
    {
        if (_network is null || _validationResult is null || !_validationResult.IsValid)
        {
            return;
        }

        var currentStage = "calculating demands";
        _calculateButton.Enabled = false;
        UseWaitCursor = true;
        _statusLabel.Text = "Calculating demands...";
        _statusLabel.Refresh();
        try
        {
            // Delegate each method to its own distributor; the UI only orchestrates the workflow.
            var distribution = _validationResult.Method switch
            {
                DemandDistributionMethod.UnitLength => UnitLengthDemandDistributor.Calculate(_network, _validationResult),
                DemandDistributionMethod.EqualDistribution => EqualDistributionDemandDistributor.Calculate(_validationResult),
                _ => throw new InvalidOperationException("The selected distribution method is not supported.")
            };

            currentStage = "writing and verifying the generated .inp file";
            _statusLabel.Text = "Writing and verifying generated .inp file...";
            _statusLabel.Refresh();
            var generatedFilePath = EpanetDemandFileWriter.WriteVerifiedCopy(_inputPathTextBox.Text, distribution);

            currentStage = "writing the detailed .txt report";
            _statusLabel.Text = "Writing detailed .txt report...";
            _statusLabel.Refresh();
            var reportFilePath = DemandGenerationReportWriter.Write(
                _inputPathTextBox.Text,
                generatedFilePath,
                _network,
                _validationResult,
                distribution);
            _validationReportTextBox.Text = distribution.BuildReport(_validationResult.FlowUnits);
            _statusLabel.Text = "Generated .inp and detailed .txt report.";
            SetValidationMessage(
                $"Generated files: {Path.GetFileName(generatedFilePath)} and {Path.GetFileName(reportFilePath)}",
                Color.DarkGreen);
        }
        catch (Exception exception) when (exception is InvalidOperationException or IOException or UnauthorizedAccessException or InvalidDataException or ArgumentException)
        {
            _validationReportTextBox.Text = $"Generation stopped while {currentStage}.\r\n{exception.Message}";
            _statusLabel.Text = $"Generation stopped while {currentStage}.";
            SetValidationMessage($"Could not complete while {currentStage}.", Color.Firebrick);
        }
        finally
        {
            UseWaitCursor = false;
            _calculateButton.Enabled = _validationResult?.IsValid == true;
        }
    }

    private void RefreshInputStatus()
    {
        SetValidationMessage(string.Empty, SystemColors.ControlText);
        if (_network is null)
        {
            return;
        }

        // Provide immediate feedback for unit mismatches and demand totals before full validation.
        var selectedUnits = _unitComboBox.SelectedItem?.ToString();
        if (selectedUnits is not null
            && _network.FlowUnits is not null
            && !selectedUnits.Equals(_network.FlowUnits, StringComparison.OrdinalIgnoreCase))
        {
            SetValidationMessage(
                $"Selected flow unit '{selectedUnits}' does not match network unit '{_network.FlowUnits}'.",
                Color.Firebrick);
            return;
        }

        if (!TryParseDemand(_totalDemandTextBox.Text, out var totalDemand))
        {
            return;
        }

        var zoneDemands = new List<double>(_zoneGrid.Rows.Count);
        foreach (DataGridViewRow row in _zoneGrid.Rows)
        {
            if (!TryParseDemand(row.Cells["Demand"].Value?.ToString(), out var zoneDemand))
            {
                return;
            }

            zoneDemands.Add(zoneDemand);
        }

        var difference = Math.Abs(zoneDemands.Sum() - totalDemand);
        var tolerance = Math.Abs(totalDemand) * DemandInputValidator.DemandMismatchToleranceFraction;
        var roundingTolerance = Math.Max(1.0, Math.Abs(totalDemand)) * 1e-9;
        if (difference <= roundingTolerance)
        {
            return;
        }

        var differenceText = totalDemand == 0
            ? "an undefined percentage of the zero total"
            : $"{difference / Math.Abs(totalDemand) * 100:F3}%";
        if (difference > tolerance)
        {
            SetValidationMessage(
                $"Zone demands differ from the total by {differenceText}; correct the discrepancy.",
                Color.Firebrick);
        }
        else
        {
            SetValidationMessage(
                $"Zone totals differ from the Network total by {difference.ToString("F3", System.Globalization.CultureInfo.InvariantCulture)} {selectedUnits ?? "unknown"}.",
                Color.DarkGoldenrod);
        }
    }

    private static bool TryParseDemand(string? text, out double value)
    {
        return (double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.CurrentCulture, out value)
                || double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out value))
            && double.IsFinite(value)
            && value >= 0;
    }

    private bool TrySaveSession()
    {
        if (_network is null || string.IsNullOrWhiteSpace(_inputPathTextBox.Text))
        {
            MessageBox.Show(this, "Select a valid EPANET input file before saving a session.", "Save session", MessageBoxButtons.OK, MessageBoxIcon.Warning);
            return false;
        }

        using var dialog = new SaveFileDialog
        {
            AddExtension = true,
            DefaultExt = "dgsession.json",
            FileName = $"{Path.GetFileNameWithoutExtension(_inputPathTextBox.Text)}.dgsession.json",
            Filter = "Demand Allocator session (*.dgsession.json)|*.dgsession.json|JSON files (*.json)|*.json",
            OverwritePrompt = true,
            Title = "Save demand generator session"
        };

        if (dialog.ShowDialog(this) != DialogResult.OK)
        {
            return false;
        }

        // Commit any active cell edit before copying values into the session record.
        _zoneGrid.EndEdit();
        var zones = _zoneGrid.Rows
            .Cast<DataGridViewRow>()
            .Select(row => new ZoneInstructionDraft(
                row.Cells["Zone"].Value?.ToString(),
                row.Cells["Demand"].Value?.ToString(),
                row.Cells["Junctions"].Value?.ToString(),
                row.Cells["KHalfPipes"].Value?.ToString(),
                row.Cells["KZeroPipes"].Value?.ToString()))
            .ToArray();

        try
        {
            DemandSessionStore.Save(
                dialog.FileName,
                new DemandSession(
                    _inputPathTextBox.Text,
                    _methodComboBox.SelectedItem?.ToString(),
                    _totalDemandTextBox.Text,
                    _unitComboBox.SelectedItem?.ToString(),
                    (int)_zoneCountSpinner.Value,
                    zones));
            _hasUnsavedChanges = false;
            SetValidationMessage("Session saved.", Color.DarkGreen);
            return true;
        }
        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException or System.Text.Json.JsonException)
        {
            SetValidationMessage("Could not save the session.", Color.Firebrick);
            MessageBox.Show(this, exception.Message, "Save session error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            return false;
        }
    }

    {
        _zoneGrid.EndEdit();
        if (_allowClose || !_hasUnsavedChanges)
        {
            return;
        }

        // The session is optional, so let the user save, discard, or cancel the close operation.
        var response = MessageBox.Show(

    }

    private void LoadSessionButton_Click(object? sender, EventArgs e)
    {
        using var dialog = new OpenFileDialog
        {
            CheckFileExists = true,
            Filter = "Demand Generator session (*.dgsession.json;*.json)|*.dgsession.json;*.json|All files (*.*)|*.*",
            Multiselect = false,
            Title = "Load demand generator session"
        };

        if (dialog.ShowDialog(this) != DialogResult.OK)
        {
            return;
        }

        try
        {
            // Validate saved paths and selections before changing the current form state.
            var session = DemandSessionStore.Load(dialog.FileName);
            if (!File.Exists(session.NetworkPath))
            {
                throw new FileNotFoundException("The input network saved in this session could not be found.", session.NetworkPath);
            }

            var network = EpanetNetworkReader.Read(session.NetworkPath);
            if (session.Method is null || !_methodComboBox.Items.Contains(session.Method))
            {
                throw new InvalidDataException("The saved distribution method is not supported.");
            }

            if (session.FlowUnits is not null && !_unitComboBox.Items.Contains(session.FlowUnits))
            {
                throw new InvalidDataException("The saved flow unit is not supported.");
            }

            _network = network;
            _inputPathTextBox.Text = session.NetworkPath;
            _totalDemandTextBox.Text = session.TotalDemand ?? string.Empty;
            _methodComboBox.SelectedItem = session.Method;
            _unitComboBox.SelectedItem = session.FlowUnits;
            _zoneCountSpinner.Value = session.ZoneCount;
            for (var index = 0; index < session.Zones.Count; index++)
            {
                var zone = session.Zones[index];
                var row = _zoneGrid.Rows[index];
                row.Cells["Zone"].Value = zone.Name;
                row.Cells["Demand"].Value = zone.Demand;
                row.Cells["Junctions"].Value = zone.Junctions;
                row.Cells["KHalfPipes"].Value = zone.KHalfPipes;
                row.Cells["KZeroPipes"].Value = zone.KZeroPipes;
            }

            ClearValidationReport();
            UpdateNetworkSummary();
            RefreshInputStatus();
            _hasUnsavedChanges = false;
        }
        catch (Exception exception) when (exception is IOException or UnauthorizedAccessException or InvalidDataException or System.Text.Json.JsonException)
        {
            SetValidationMessage("Could not load the saved session.", Color.Firebrick);
            MessageBox.Show(this, exception.Message, "Load session error", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }

    private void UpdateNetworkSummary()
    {
        if (_network is null)
        {
            _statusLabel.Text = "No input file selected";
            return;
        }

        var unitText = _network.FlowUnits is null ? "unknown units" : _network.FlowUnits;
        _statusLabel.Text = $"{_network.JunctionIds.Count} junctions | {_network.ReservoirIds.Count} reservoirs | " +
            $"{_network.TankIds.Count} tanks | {_network.Pipes.Count} pipes | {_network.PumpCount} pumps | " +
            $"{_network.ValveCount} valves | {unitText}";
    }

    private void ValidateButton_Click(object? sender, EventArgs e)
    {
        if (_network is null)
        {
            _validationResult = null;

            _validationReportTextBox.Text = "Validation Error. See detail validation report.\r\nValidation errors\r\n- Select a valid EPANET input file first.";
            SetValidationMessage("Validation Error. See detail validation report.", Color.Firebrick);
            return;
        }

        // Commit the active grid cell so validation sees the latest value the user entered.
        _zoneGrid.EndEdit();
        var zoneDrafts = _zoneGrid.Rows
            .Cast<DataGridViewRow>()
            .Select(row => new ZoneInstructionDraft(
                row.Cells["Zone"].Value?.ToString(),
                row.Cells["Demand"].Value?.ToString(),
                row.Cells["Junctions"].Value?.ToString(),
                row.Cells["KHalfPipes"].Value?.ToString(),
                row.Cells["KZeroPipes"].Value?.ToString()))
            .ToArray();

        var result = DemandInputValidator.Validate(
            _network,
            GetSelectedDistributionMethod(),
            _totalDemandTextBox.Text,
            _unitComboBox.SelectedItem?.ToString(),
            zoneDrafts);
        _validationResult = result;
        _calculateButton.Enabled = result.IsValid;
        _validationReportTextBox.Text = result.BuildReport();

        {
            SetValidationMessage("Validation Error. See detail validation report.", Color.Firebrick);

        {
            SetValidationMessage("Validation passed with a demand-rounding warning; review details above.", Color.DarkGoldenrod);
        }
        else
        {
            SetValidationMessage("Validation passed. No demands have been calculated or written.", Color.DarkGreen);
        }
    }

    private void CalculateButton_Click(object? sender, EventArgs e)
    {
        if (_network is null || _validationResult is null || !_validationResult.IsValid)
        {
            return;
        }

        var currentStage = "calculating demands";
        _calculateButton.Enabled = false;
        UseWaitCursor = true;
        _statusLabel.Text = "Calculating demands...";
        _statusLabel.Refresh();
        try
        {
            // Delegate each method to its own distributor; the UI only orchestrates the workflow.
            var distribution = _validationResult.Method switch
            {
                DemandDistributionMethod.UnitLength => UnitLengthDemandDistributor.Calculate(_network, _validationResult),
                DemandDistributionMethod.EqualDistribution => EqualDistributionDemandDistributor.Calculate(_validationResult),
                _ => throw new InvalidOperationException("The selected distribution method is not supported.")
            };

            currentStage = "writing and verifying the generated .inp file";
            _statusLabel.Text = "Writing and verifying generated .inp file...";
            _statusLabel.Refresh();
            var generatedFilePath = EpanetDemandFileWriter.WriteVerifiedCopy(_inputPathTextBox.Text, distribution);

            currentStage = "writing the detailed .txt report";
            _statusLabel.Text = "Writing detailed .txt report...";
            _statusLabel.Refresh();
            var reportFilePath = DemandGenerationReportWriter.Write(
                _inputPathTextBox.Text,
                generatedFilePath,
                _network,
                _validationResult,
                distribution);
            _validationReportTextBox.Text = distribution.BuildReport(_validationResult.FlowUnits);
            _statusLabel.Text = "Generated .inp and detailed .txt report.";
            SetValidationMessage(
                $"Generated files: {Path.GetFileName(generatedFilePath)} and {Path.GetFileName(reportFilePath)}",
                Color.DarkGreen);
        }
        catch (Exception exception) when (exception is InvalidOperationException or IOException or UnauthorizedAccessException or InvalidDataException or ArgumentException)
        {
            _validationReportTextBox.Text = $"Generation stopped while {currentStage}.\r\n{exception.Message}";
            _statusLabel.Text = $"Generation stopped while {currentStage}.";
            SetValidationMessage($"Could not complete while {currentStage}.", Color.Firebrick);
        }
        finally
        {
            UseWaitCursor = false;
            _calculateButton.Enabled = _validationResult?.IsValid == true;
        }
    }

    private void RefreshInputStatus()
    {
        SetValidationMessage(string.Empty, SystemColors.ControlText);
        if (_network is null)
        {
            return;
        }

        // Provide immediate feedback for unit mismatches and demand totals before full validation.
        var selectedUnits = _unitComboBox.SelectedItem?.ToString();
        if (selectedUnits is not null
            && _network.FlowUnits is not null
            && !selectedUnits.Equals(_network.FlowUnits, StringComparison.OrdinalIgnoreCase))
        {
            SetValidationMessage(
                $"Selected flow unit '{selectedUnits}' does not match network unit '{_network.FlowUnits}'.",
                Color.Firebrick);
            return;
        }

        if (!TryParseDemand(_totalDemandTextBox.Text, out var totalDemand))
        {
            return;
        }

        var zoneDemands = new List<double>(_zoneGrid.Rows.Count);
        foreach (DataGridViewRow row in _zoneGrid.Rows)
        {
            if (!TryParseDemand(row.Cells["Demand"].Value?.ToString(), out var zoneDemand))
            {
                return;
            }

            zoneDemands.Add(zoneDemand);
        }

        var difference = Math.Abs(zoneDemands.Sum() - totalDemand);
        var tolerance = Math.Abs(totalDemand) * DemandInputValidator.DemandMismatchToleranceFraction;
        var roundingTolerance = Math.Max(1.0, Math.Abs(totalDemand)) * 1e-9;
        if (difference <= roundingTolerance)
        {
            return;
        }

        var differenceText = totalDemand == 0
            ? "an undefined percentage of the zero total"
            : $"{difference / Math.Abs(totalDemand) * 100:F3}%";
        if (difference > tolerance)
        {
            SetValidationMessage(
                $"Zone demands differ from the total by {differenceText}; correct the discrepancy.",
                Color.Firebrick);
        }
        else
        {
            SetValidationMessage(
                $"Zone totals differ from the Network total by {difference.ToString("F3", System.Globalization.CultureInfo.InvariantCulture)} {selectedUnits ?? "unknown"}.",
                Color.DarkGoldenrod);
        }
    }

    private static bool TryParseDemand(string? text, out double value)
    {
        return (double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.CurrentCulture, out value)
                || double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out value))
            && double.IsFinite(value)
            && value >= 0;
    }

# EPANET-JDA (EPANET Junction Demand Allocator)

EPANET-JDA is a Windows-based utility which calculates junction base demands of an EPANET 2.2 hydraulic network model. It supports two standard demand allocation methodologies: **Unit-Length** and **Equal-Distribution** (refer later sections for detail). The application operates strictly as a pre-processor; it does not execute EPANET hydraulic simulations and does not require the EPANET simulation engine installed on the host machine.

The program reads a user-supplied input (.inp) file of EPANET 2.2, computes junction demands according to user-defined zones and weighting parameters, and writes a separate, verified input (.inp) file. The source file is never overwritten or altered.

### 2. System Requirements and Installation

**Operating System:** Windows 10 or Windows 11.
**Runtime:** Standalone desktop executable (no external installer or runtime setup wizard required).

See the epanetjdahelp.pdf (user guide) file for installation and operation instructions, allocation rules, and troubleshooting.

## Run the standalone Windows application

**Package Archive (epanetjda.zip):** Contains epanetjda.exe, epanetjdahelp.pdf (user guide), and one sample input (.inp) file of EPANET 2.2. 

Download the Windows x64 release ZIP and extract it. Run **epanetjda.exe** from the extracted folder. The packaged executable is self-contained, so no separate .NET runtime installation is required.

## Publish a standalone x64 executable

To publish a self-contained, single-file Windows x64 application:

```powershell
dotnet publish .\Demand_Gen.csproj -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true -p:IncludeNativeLibrariesForSelfExtract=true
```

The publish output is written to `bin\Release\net10.0-windows\win-x64\publish`. Distribute the executable together with the published help PDF and sample network file.

Review the generated input file before using it in EPANET.

using System.Globalization;
using System.Text;

namespace Demand_Gen;

public sealed record JunctionDemandAllocation(
    string JunctionId,
    string ZoneName,
    double AllocationWeight,
    double Demand);

// Groups the computed junction demands and their requested total for one zone.
public sealed record ZoneDemandAllocation(
    string ZoneName,
    double RequestedDemand,
    IReadOnlyList<JunctionDemandAllocation> Junctions);

// Holds the complete allocation result independently of the UI or file writer.
public sealed record DemandDistributionResult(
    DemandDistributionMethod Method,
    IReadOnlyList<ZoneDemandAllocation> Zones)
{
    // Produces a concise summary for the application's validation/report panel.
    public string BuildReport(string? flowUnits)
    {
        var report = new StringBuilder();
        report.AppendLine(Method == DemandDistributionMethod.UnitLength
            ? "Unit-Length calculation:"
            : "Equal-Distribution calculation:");

        // Flatten zone results so network-wide totals and junction counts include every zone.
        var junctions = Zones.SelectMany(zone => zone.Junctions).ToArray();
        var totalDemand = junctions.Sum(junction => junction.Demand);
        var junctionsWithDemand = junctions.Count(junction => junction.Demand > 0);
        var junctionsWithZeroDemand = junctions.Length - junctionsWithDemand;
        report.AppendLine($"Total demand: {FormatNumber(totalDemand)} {flowUnits ?? "unknown"}");
        report.AppendLine($"Junctions with demand: {junctionsWithDemand}");
        report.AppendLine($"Junctions with zero demand: {junctionsWithZeroDemand}");
        return report.ToString();
    }

    private static string FormatNumber(double value) =>
        value.ToString("F4", CultureInfo.InvariantCulture);
}

public static class UnitLengthDemandDistributor
{
    /// <summary>
    /// Allocates each zone's demand in proportion to K-weighted pipe length at its junctions.
    /// </summary>
    public static DemandDistributionResult Calculate(
        EpanetNetworkSummary network,
        DemandValidationResult validation)
    {
        ArgumentNullException.ThrowIfNull(network);
        ArgumentNullException.ThrowIfNull(validation);
        if (!validation.IsValid)
        {
            throw new InvalidOperationException("Demand instructions must pass validation before calculation.");
        }

        if (validation.Method != DemandDistributionMethod.UnitLength)
        {
            throw new InvalidOperationException("Unit-Length calculation requires Unit-Length validation.");
        }

        // Resolve junction ownership and explicit K overrides once, using EPANET's case-insensitive IDs.
        var zonesByJunction = new Dictionary<string, ParsedZoneInstruction>(StringComparer.OrdinalIgnoreCase);
        var kByPipe = new Dictionary<string, double>(StringComparer.OrdinalIgnoreCase);
        foreach (var zone in validation.Zones)
        {
            foreach (var junctionId in zone.Junctions)
            {
                if (!zonesByJunction.TryAdd(junctionId, zone))
                {
                    throw new InvalidOperationException($"Junction '{junctionId}' is assigned to more than one zone.");
                }
            }

            foreach (var pipeId in zone.KHalfPipes)
            {
                // TryAdd also guards against duplicate assignments if this method is called without valid input.
                if (!kByPipe.TryAdd(pipeId, 0.5))
                {
                    throw new InvalidOperationException($"Pipe '{pipeId}' has more than one K-factor assignment.");
                }
            }

            foreach (var pipeId in zone.KZeroPipes)
            {
                if (!kByPipe.TryAdd(pipeId, 0.0))
                {
                    throw new InvalidOperationException($"Pipe '{pipeId}' has more than one K-factor assignment.");
                }
            }
        }

        var pipeWeights = network.JunctionIds.ToDictionary(
            id => id,
            _ => 0.0,
            StringComparer.OrdinalIgnoreCase);

        // Accumulate each pipe's effective length at its junction endpoints.
        // Junction-to-junction pipes are split evenly; a reservoir/tank endpoint receives no share.
        foreach (var pipe in network.Pipes)
        {
            var k = kByPipe.TryGetValue(pipe.Id, out var assignedK) ? assignedK : 1.0;
            var effectiveLength = pipe.Length * k;
            if (!double.IsFinite(effectiveLength))
            {
                throw new InvalidOperationException($"Pipe '{pipe.Id}' produces a non-finite K-weighted length.");
            }

            var hasStartJunction = zonesByJunction.ContainsKey(pipe.StartNodeId);
            var hasEndJunction = zonesByJunction.ContainsKey(pipe.EndNodeId);
            if (hasStartJunction && hasEndJunction)
            {
                var endpointWeight = effectiveLength / 2.0;
                AddWeight(pipeWeights, pipe.StartNodeId, endpointWeight);
                AddWeight(pipeWeights, pipe.EndNodeId, endpointWeight);
            }
            else if (hasStartJunction)
            {
                AddWeight(pipeWeights, pipe.StartNodeId, effectiveLength);
            }
            else if (hasEndJunction)
            {
                AddWeight(pipeWeights, pipe.EndNodeId, effectiveLength);
            }
        }

        var zoneAllocations = new List<ZoneDemandAllocation>(validation.Zones.Count);
        foreach (var zone in validation.Zones)
        {
            var requestedDemand = zone.Demand
                ?? throw new InvalidOperationException($"Zone '{zone.Name}' has no validated demand.");
            var includedIds = zone.Junctions;
            var totalIncludedWeight = includedIds.Sum(id => pipeWeights[id]);
            if (!double.IsFinite(totalIncludedWeight))
            {
                throw new InvalidOperationException($"Zone '{zone.Name}' has a non-finite total pipe weight.");
            }

            if (requestedDemand > 0 && totalIncludedWeight <= 0)
            {
                throw new InvalidOperationException(
                    $"Zone '{zone.Name}' has no positive K-weighted pipe length for its demand.");
            }

            // A zero-demand zone remains valid even when all its junction weights are zero.
            var demands = zone.Junctions.ToDictionary(
                id => id,
                _ => 0.0,
                StringComparer.OrdinalIgnoreCase);
            if (requestedDemand > 0)
            {
                foreach (var junctionId in includedIds)
                {
                    demands[junctionId] = requestedDemand * (pipeWeights[junctionId] / totalIncludedWeight);
                }

                // Put floating-point rounding residue on the highest-weight junction to preserve the zone total.
                var allocated = includedIds.Sum(id => demands[id]);
                var correctionJunction = includedIds.MaxBy(id => pipeWeights[id]);
                demands[correctionJunction!] += requestedDemand - allocated;
            }

            var junctionAllocations = zone.Junctions
                .Select(id => new JunctionDemandAllocation(id, zone.Name, pipeWeights[id], demands[id]))
                .ToArray();
            zoneAllocations.Add(new ZoneDemandAllocation(zone.Name, requestedDemand, junctionAllocations));
        }

        return new DemandDistributionResult(DemandDistributionMethod.UnitLength, zoneAllocations);
    }

    private static void AddWeight(IDictionary<string, double> pipeWeights, string junctionId, double weight)
    {
        pipeWeights[junctionId] += weight;
        // Detect overflow at the point it enters the accumulated junction weight.
        if (!double.IsFinite(pipeWeights[junctionId]))
        {
            throw new InvalidOperationException($"Junction '{junctionId}' has a non-finite pipe weight.");
        }
    }
